package Ejercicio2;

import java.io.*;
import java.util.*;

public class AppMain {

    static String ARCHIVO_MED = "medicos.txt";
    static String ARCHIVO_CONS = "consultas.txt";

    public static void main(String[] args) throws Exception {

        // --- a) Alta de 3 médicos ---
        altaMedico(new Medico(1, "Juan", "Perez", 5));
        altaMedico(new Medico(2, "Luis", "Lopez", 7));
        altaMedico(new Medico(3, "Ana", "Vega", 10));

        // --- a) Alta de 9 consultas ---
        altaConsulta(new Consulta(101, "Mario", "Santos", 1, 25, "Diciembre", 2024));
        altaConsulta(new Consulta(102, "Roberto", "Luna", 2, 24, "Diciembre", 2024));
        altaConsulta(new Consulta(103, "Jose", "Campos", 3, 31, "Diciembre", 2024));
        altaConsulta(new Consulta(104, "Ana", "Rojas", 1, 1, "Enero", 2025));
        altaConsulta(new Consulta(105, "Lucia", "Diaz", 2, 15, "Febrero", 2025));
        altaConsulta(new Consulta(106, "Carlos", "Gomez", 3, 10, "Marzo", 2025));
        altaConsulta(new Consulta(107, "Sofia", "Torres", 1, 5, "Abril", 2025));
        altaConsulta(new Consulta(108, "Miguel", "Perez", 2, 20, "Mayo", 2025));
        altaConsulta(new Consulta(109, "Elena", "Lopez", 3, 30, "Junio", 2025));

        System.out.println("----- Médicos -----");
        imprimirArchivo(ARCHIVO_MED);

        System.out.println("----- Consultas -----");
        imprimirArchivo(ARCHIVO_CONS);

        // --- b) Baja médico y sus consultas ---
        bajaMedico("Luis", "Lopez"); // Ejemplo
        System.out.println("\nMédico Luis Lopez eliminado (y sus consultas).");
        System.out.println("----- Médicos -----");
        imprimirArchivo(ARCHIVO_MED);
        System.out.println("----- Consultas -----");
        imprimirArchivo(ARCHIVO_CONS);

        // --- c) Cambiar consultas de Navidad/Año Nuevo ---
        cambiarConsultasNavidad();
        System.out.println("\nConsultas de Navidad/Año Nuevo modificadas.");
        System.out.println("----- Consultas -----");
        imprimirArchivo(ARCHIVO_CONS);

        // --- Opcional: mostrar pacientes en mi cumpleaños ---
        System.out.println("\nPacientes en mi cumpleaños (25 Diciembre):");
        mostrarPacientesPorDia(25, "Diciembre");

    }

    // --- Métodos principales ---
    static void altaMedico(Medico m) throws Exception {
        FileWriter fw = new FileWriter(ARCHIVO_MED, true);
        fw.write(m.toString() + "\n");
        fw.close();
    }

    static void altaConsulta(Consulta c) throws Exception {
        FileWriter fw = new FileWriter(ARCHIVO_CONS, true);
        fw.write(c.toString() + "\n");
        fw.close();
    }

    static void bajaMedico(String nombre, String apellido) throws Exception {
        File input = new File(ARCHIVO_MED);
        File temp = new File("temp.txt");

        BufferedReader br = new BufferedReader(new FileReader(input));
        PrintWriter pw = new PrintWriter(new FileWriter(temp));

        String linea;
        while ((linea = br.readLine()) != null) {
            String[] p = linea.split(";");
            if (!(p[1].equals(nombre) && p[2].equals(apellido))) {
                pw.println(linea);
            }
        }
        br.close();
        pw.close();

        input.delete();
        temp.renameTo(input);

        // Eliminar consultas del médico
        bajaConsultasPorMedico(nombre, apellido);
    }

    static void bajaConsultasPorMedico(String nombre, String apellido) throws Exception {
        int id = obtenerIdMedico(nombre, apellido);
        if (id == -1) return;

        File input = new File(ARCHIVO_CONS);
        File temp = new File("temp2.txt");

        BufferedReader br = new BufferedReader(new FileReader(input));
        PrintWriter pw = new PrintWriter(new FileWriter(temp));

        String linea;
        while ((linea = br.readLine()) != null) {
            String[] p = linea.split(";");
            if (Integer.parseInt(p[3]) != id) {
                pw.println(linea);
            }
        }
        br.close();
        pw.close();

        input.delete();
        temp.renameTo(input);
    }

    static int obtenerIdMedico(String nombre, String apellido) throws Exception {
        BufferedReader br = new BufferedReader(new FileReader(ARCHIVO_MED));
        String linea;
        while ((linea = br.readLine()) != null) {
            String[] p = linea.split(";");
            if (p[1].equals(nombre) && p[2].equals(apellido)) {
                br.close();
                return Integer.parseInt(p[0]);
            }
        }
        br.close();
        return -1;
    }

    static void cambiarConsultasNavidad() throws Exception {
        File input = new File(ARCHIVO_CONS);
        File temp = new File("temp3.txt");

        BufferedReader br = new BufferedReader(new FileReader(input));
        PrintWriter pw = new PrintWriter(new FileWriter(temp));

        String linea;
        while ((linea = br.readLine()) != null) {
            String[] p = linea.split(";");

            int dia = Integer.parseInt(p[4]);
            String mes = p[5];

            if ((dia == 25 && mes.equals("Diciembre")) || (dia == 1 && mes.equals("Enero"))) {
                p[4] = "2"; // Cambiar día
                linea = String.join(";", p);
            }

            pw.println(linea);
        }
        br.close();
        pw.close();

        input.delete();
        temp.renameTo(input);
    }

    static void mostrarPacientesPorDia(int dia, String mes) throws Exception {
        BufferedReader br = new BufferedReader(new FileReader(ARCHIVO_CONS));
        String linea;
        while ((linea = br.readLine()) != null) {
            String[] p = linea.split(";");
            int d = Integer.parseInt(p[4]);
            String m = p[5];
            if (d == dia && m.equals(mes)) {
                System.out.println(p[1] + " " + p[2] + " (ID Med: " + p[3] + ")");
            }
        }
        br.close();
    }

    static void imprimirArchivo(String archivo) throws Exception {
        File f = new File(archivo);
        if (!f.exists()) {
            System.out.println("Archivo vacío o no existe.");
            return;
        }
        BufferedReader br = new BufferedReader(new FileReader(f));
        String linea;
        while ((linea = br.readLine()) != null) {
            System.out.println(linea);
        }
        br.close();
    }
}
